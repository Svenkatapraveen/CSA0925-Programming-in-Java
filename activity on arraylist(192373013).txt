import java.util.ArrayList;
class Student{
    String name;
    int regNo;
    int marks;
    Student(String name, int regNo, int marks){
        this.name = name;
        this.regNo = regNo;
        this.marks = marks;
    }
    void display(){
        System.out.println("Name : " + name);
        System.out.println("Reg No : " + regNo);
        System.out.println("Marks : " + marks);
        if(marks >= 50){
            System.out.println("Result : pass");
        } else {
            System.out.println("Result : fail");
        }
        System.out.println();
    }
}
public class Main {
    public static void main(String[] args){
        ArrayList<Student> students = new ArrayList<>();
        students.add(new Student("ravi", 10112, 50));
        students.add(new Student("naveen", 10234, 36));
        students.add(new Student("rajesh", 10356, 55));
        students.add(new Student("rahul", 10342, 26));
        for(Student s : students){
            s.display();
        }
    }
}
